# Bootstrap New AWS Account for PrivacyReady Platform

# 1. Create the Remote State Infrastructure
# First, use the AWS CLI to manually create the S3 bucket and DynamoDB table that Terraform will use for state locking.
# (If `privacyready-terraform-state` is globally taken, change it here and in `terraform/versions.tf`):

# Create the S3 bucket for state storage
aws s3api create-bucket \
  --bucket privacyready-terraform-state \
  --region ap-southeast-1 \
  --create-bucket-configuration LocationConstraint=ap-southeast-1

# Enable versioning on the bucket (critical for state backups)
aws s3api put-bucket-versioning \
  --bucket privacyready-terraform-state \
  --versioning-configuration Status=Enabled

# Create the DynamoDB table for state locking
aws dynamodb create-table \
  --table-name privacyready-terraform-locks \
  --attribute-definitions AttributeName=LockID,AttributeType=S \
  --key-schema AttributeName=LockID,KeyType=HASH \
  --billing-mode PAY_PER_REQUEST \
  --region ap-southeast-1


# 2. Initialize Terraform
# Now that the remote backend infrastructure exists, you can initialize Terraform.
# Note: If you have a local `terraform.tfstate`, delete it or allow Terraform to migrate it.
cd terraform
terraform init


# 3. Build the ECR Repositories First
# Because your ECS clusters depend on Docker images existing in ECR, you must create the container registries *before* deploying the rest of the infrastructure.
terraform apply -target=aws_ecr_repository.api -target=aws_ecr_repository.scanner -target=aws_ecr_repository.dsr


# 4. Build and Push Your Docker Images
# Log into ECR and push your backend and scanner images to the new repositories.
# REPLACE <YOUR_ACCOUNT_ID> with your actual AWS Account ID.

export AWS_ACCOUNT_ID="<YOUR_ACCOUNT_ID>"

# Log in to ECR
aws ecr get-login-password --region ap-southeast-1 | docker login --username AWS --password-stdin ${AWS_ACCOUNT_ID}.dkr.ecr.ap-southeast-1.amazonaws.com

# Build and push API
cd ../services/api
docker build -t privacyready-api .
docker tag privacyready-api:latest ${AWS_ACCOUNT_ID}.dkr.ecr.ap-southeast-1.amazonaws.com/privacyready-api:latest
docker push ${AWS_ACCOUNT_ID}.dkr.ecr.ap-southeast-1.amazonaws.com/privacyready-api:latest

# Build and push Scanner
cd ../scanner
docker build -t privacyready-scanner .
docker tag privacyready-scanner:latest ${AWS_ACCOUNT_ID}.dkr.ecr.ap-southeast-1.amazonaws.com/privacyready-scanner:latest
docker push ${AWS_ACCOUNT_ID}.dkr.ecr.ap-southeast-1.amazonaws.com/privacyready-scanner:latest


# 5. Deploy the Full Environment
# With your state safely in S3 and your images loaded into ECR, you can now safely apply the entire infrastructure stack:
cd ../../terraform
terraform apply

# 6. Post-Deployment
# Run the following to get your newly generated Route 53 Nameservers, then update your domain registrar:
terraform output domain_nameservers
